from flask import Flask, render_template from datetime import datetime app = Flask(__name__) @app.route('/') def index(): now = datetime.now() today_date = now.strftime("%d.%m.%Y") days = ["Понеділок", "Вівторок", "Середа", "Четвер", "П'ятниця", "Субота", "Неділя"] day_name = days[now.weekday()] return render_template('index.html', today_date=today_date, day_name=day_name) if __name__ == '__main__': app.run(host='0.0.0.0', port=5000) ├── requirements.txt │ ├── Flask==3.0.2 │ └── gunicorn==21.2.0 ├── Procfile │ └── web: gunicorn app:app └── .gitignore __pycache__/ *.pyc venv/ .env
